);

CREATE TABLE notes (

    content    TEXT         NOT NULL,
    type_id    INT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (type_id) REFERENCES note_types(id)

package com.codegym.model;

/**
 * Note — Interface (Strategy) định nghĩa cách quản lý ghi chú.
 *
 * Có 2 chiến lược (concrete strategies):
 *  - NoteDB   : lưu vào CSDL MySQL
 *  - NoteFile : lưu vào file văn bản
 *
 * Nhờ Strategy Pattern, NoteManagement có thể đổi chiến lược
 * lưu trữ mà không cần sửa code nghiệp vụ.
 */
public interface Note {

    void setId(int id);
    int getId();

    void setTitle(String title);
    void setContent(String content);
    void setTypeId(int typeId);

    String getTitle();
    String getContent();
    String getType();

    /** Lưu ghi chú (INSERT hoặc UPDATE tùy impl). */
    void save();

    /** Xóa ghi chú theo id. */
    void delete();
}

package com.codegym.model;

import com.codegym.dao.DBConnection;

import java.sql.*;

/**
 * NoteDB — Chiến lược lưu ghi chú vào CSDL MySQL.
 */
public class NoteDB implements Note {

    private int id;
    private String title;
    private String content;
    private int typeId;
    private String typeName;

    public NoteDB() {}

    public NoteDB(int id, String title, String content,
                  int typeId, String typeName) {
        this.id = id;
        this.title = title;
        this.content = content;
        this.typeId = typeId;
        this.typeName = typeName;
    }

    public NoteDB(String title, String content, int typeId) {
        this.title = title;
        this.content = content;
        this.typeId = typeId;
    }

    @Override public void setId(int id) { this.id = id; }
    @Override public int getId() { return id; }

    @Override public void setTitle(String title) { this.title = title; }
    @Override public void setContent(String content) { this.content = content; }
    @Override public void setTypeId(int typeId) { this.typeId = typeId; }

    @Override public String getTitle() { return title; }
    @Override public String getContent() { return content; }
    @Override public String getType() { return typeName; }

    public int getTypeId() { return typeId; }

    // ============================================================
    // SAVE — INSERT hoặc UPDATE tùy có id hay chưa
    // ============================================================
    @Override
    public void save() {
        if (id > 0) {
            update();
        } else {
            insert();
        }
    }

    private void insert() {
        String sql = "INSERT INTO notes (title, content, type_id) VALUES (?, ?, ?)";
        try (Connection c = DBConnection.getConnection();
             PreparedStatement ps = c.prepareStatement(
                 sql, Statement.RETURN_GENERATED_KEYS)) {
            ps.setString(1, title);
            ps.setString(2, content);
            ps.setInt(3, typeId);
            ps.executeUpdate();

            ResultSet rs = ps.getGeneratedKeys();
            if (rs.next()) id = rs.getInt(1);
        } catch (SQLException e) { e.printStackTrace(); }
    }

    private void update() {
        String sql = "UPDATE notes SET title=?, content=?, type_id=? WHERE id=?";
        try (Connection c = DBConnection.getConnection();
             PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setString(1, title);
            ps.setString(2, content);
            ps.setInt(3, typeId);
            ps.setInt(4, id);
            ps.executeUpdate();
        } catch (SQLException e) { e.printStackTrace(); }
    }

    // ============================================================
    // DELETE
    // ============================================================
    @Override
    public void delete() {
        String sql = "DELETE FROM notes WHERE id = ?";
        try (Connection c = DBConnection.getConnection();
             PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setInt(1, id);
            ps.executeUpdate();
        } catch (SQLException e) { e.printStackTrace(); }
    }
}

package com.codegym.model;

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.ArrayList;
import java.util.List;

/**
 * NoteFile — Chiến lược lưu ghi chú vào file văn bản.
 *
 * Định dạng file notes.txt (mỗi dòng 1 ghi chú):
 *   id|title|typeId|content
 *
 * Ví dụ:
 *   1|Họp nhóm|1|Nội dung cuộc họp...
 *   2|Mua sữa|2|Mua 2 hộp sữa tươi
 */
public class NoteFile implements Note {

    private static final String FILE_PATH = "notes.txt";
    private static final String SEPARATOR = "|";

    private int id;
    private String title;
    private String content;
    private int typeId;
    private String typeName;

    public NoteFile() {}

    public NoteFile(int id, String title, String content, int typeId) {
        this.id = id;
        this.title = title;
        this.content = content;
        this.typeId = typeId;
    }

    @Override public void setId(int id) { this.id = id; }
    @Override public int getId() { return id; }

    @Override public void setTitle(String title) { this.title = title; }
    @Override public void setContent(String content) { this.content = content; }
    @Override public void setTypeId(int typeId) { this.typeId = typeId; }

    @Override public String getTitle() { return title; }
    @Override public String getContent() { return content; }
    @Override public String getType() { return typeName; }

    public int getTypeId() { return typeId; }
    public void setTypeName(String typeName) { this.typeName = typeName; }

    // ============================================================
    // SAVE — Ghi vào file
    // ============================================================
    @Override
    public void save() {
        List<String> lines = readAllLines();

        // Tìm id lớn nhất để sinh id mới
        if (id <= 0) {
            int maxId = 0;
            for (String line : lines) {
                String[] parts = line.split("\\|", 4);
                if (parts.length >= 1) {
                    try { maxId = Math.max(maxId, Integer.parseInt(parts[0])); }
                    catch (NumberFormatException ignored) {}
                }
            }
            id = maxId + 1;
        }

        String newLine = id + SEPARATOR + title + SEPARATOR + typeId + SEPARATOR + content;

        // Ghi đè dòng có cùng id (nếu update) hoặc thêm mới
        boolean updated = false;
        for (int i = 0; i < lines.size(); i++) {
            String[] parts = lines.get(i).split("\\|", 4);
            if (parts.length >= 1 && String.valueOf(id).equals(parts[0])) {
                lines.set(i, newLine);
                updated = true;
                break;
            }
        }
        if (!updated) lines.add(newLine);

        writeAllLines(lines);
    }

    // ============================================================
    // DELETE — Xóa khỏi file
    // ============================================================
    @Override
    public void delete() {
        List<String> lines = readAllLines();
        lines.removeIf(line -> {
            String[] parts = line.split("\\|", 4);
            return parts.length >= 1 && String.valueOf(id).equals(parts[0]);
        });
        writeAllLines(lines);
    }

    // ============================================================
    // ĐỌC / GHI FILE
    // ============================================================
    private List<String> readAllLines() {
        File file = new File(FILE_PATH);
        if (!file.exists()) return new ArrayList<>();

        try {
            return Files.readAllLines(file.toPath(), StandardCharsets.UTF_8);
        } catch (IOException e) {
            e.printStackTrace();
            return new ArrayList<>();
        }
    }

    private void writeAllLines(List<String> lines) {
        try {
            Files.write(Paths.get(FILE_PATH), lines, StandardCharsets.UTF_8,
                        StandardOpenOption.CREATE,
                        StandardOpenOption.TRUNCATE_EXISTING,
                        StandardOpenOption.WRITE);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    /**
     * Đọc tất cả ghi chú từ file và trả về List<NoteFile>.
     */
    public static List<NoteFile> findAll() {
        List<NoteFile> result = new ArrayList<>();
        File file = new File(FILE_PATH);
        if (!file.exists()) return result;

        try {
            List<String> lines = Files.readAllLines(file.toPath(), StandardCharsets.UTF_8);
            for (String line : lines) {
                String[] parts = line.split("\\|", 4);
                if (parts.length >= 4) {
                    NoteFile n = new NoteFile(
                        Integer.parseInt(parts[0]),
                        parts[1],
                        parts[3],
                        Integer.parseInt(parts[2])
                    );
                    result.add(n);
                }
            }
        } catch (IOException e) { e.printStackTrace(); }
        return result;
    }

    /**
     * Tìm ghi chú theo id trong file.
     */
    public static NoteFile findById(int id) {
        for (NoteFile n : findAll()) {
            if (n.getId() == id) return n;
        }
        return null;
    }
}

package com.codegym.service;

import com.codegym.dao.DBConnection;
import com.codegym.model.Note;
import com.codegym.model.NoteDB;
import com.codegym.model.NoteFile;
import com.codegym.model.NoteType;

import java.sql.*;
import java.util.ArrayList;
import java.util.List;

/**
 * NoteManagement — Lớp quản lý ghi chú sử dụng Strategy Pattern.
 *
 * Có thể chuyển đổi giữa 2 chiến lược lưu trữ:
 *  - "DB"   → NoteDB
 *  - "FILE" → NoteFile
 *
 * Sử dụng: changeNoteStore("FILE") để đổi sang lưu file.
 */
public class NoteManagement {

    /** Loại store hiện tại: "DB" hoặc "FILE" */
    private String storeType = "DB";

    /** Note hiện tại (dùng cho save/delete khi đã setTitle/setContent) */
    private Note note;

    public NoteManagement() {
        changeNoteStore("DB");
    }

    // ============================================================
    // ⭐ CHIẾN LƯỢC — Đổi store type
    // ============================================================
    public void changeNoteStore(String storeType) {
        this.storeType = storeType;
        if ("FILE".equalsIgnoreCase(storeType)) {
            this.note = new NoteFile();
        } else {
            this.note = new NoteDB();
        }
    }

    public String getStoreType() { return storeType; }

    // ============================================================
    // SỬ DỤNG NOTE HIỆN TẠI (Strategy được chọn)
    // ============================================================

    /** Thêm note — sử dụng note strategy hiện tại. */
    public void addNote(String title, String content, int typeId) {
        note.setTitle(title);
        note.setContent(content);
        note.setTypeId(typeId);
        note.save();
    }

    /** Xóa note theo id — sử dụng note strategy hiện tại. */
    public void addNote(int id) {
        note.setId(id);
        note.delete();
    }

    // ============================================================
    // TÌM KIẾM — trả về List<Note> tùy theo storeType
    // ============================================================
    public List<Note> searchNotes(String keyword) {
        if ("FILE".equalsIgnoreCase(storeType)) {
            return searchInFile(keyword);
        }
        return searchInDB(keyword);
    }

    private List<Note> searchInDB(String keyword) {
        List<Note> result = new ArrayList<>();
        String sql =
            "SELECT n.id, n.title, n.content, n.type_id, t.name AS type_name " +
            "FROM notes n LEFT JOIN note_types t ON n.type_id = t.id " +
            "WHERE n.title LIKE ? OR n.content LIKE ? " +
            "ORDER BY n.id DESC";

        try (Connection c = DBConnection.getConnection();
             PreparedStatement ps = c.prepareStatement(sql)) {
            String kw = "%" + (keyword == null ? "" : keyword.trim()) + "%";
            ps.setString(1, kw);
            ps.setString(2, kw);
            ResultSet rs = ps.executeQuery();
            while (rs.next()) {
                result.add(new NoteDB(
                    rs.getInt("id"),
                    rs.getString("title"),
                    rs.getString("content"),
                    rs.getInt("type_id"),
                    rs.getString("type_name")
                ));
            }
        } catch (SQLException e) { e.printStackTrace(); }
        return result;
    }

    private List<Note> searchInFile(String keyword) {
        List<Note> result = new ArrayList<>();
        String kw = (keyword == null) ? "" : keyword.trim().toLowerCase();
        for (NoteFile n : NoteFile.findAll()) {
            if (n.getTitle().toLowerCase().contains(kw)
                    || n.getContent().toLowerCase().contains(kw)) {
                result.add(n);
            }
        }
        return result;
    }

    // ============================================================
    // LẤY 1 NOTE THEO ID
    // ============================================================
    public Note getNoteById(int id) {
        if ("FILE".equalsIgnoreCase(storeType)) {
            return NoteFile.findById(id);
        }
        String sql =
            "SELECT n.id, n.title, n.content, n.type_id, t.name AS type_name " +
            "FROM notes n LEFT JOIN note_types t ON n.type_id = t.id " +
            "WHERE n.id = ?";
        try (Connection c = DBConnection.getConnection();
             PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setInt(1, id);
            ResultSet rs = ps.executeQuery();
            if (rs.next()) {
                return new NoteDB(
                    rs.getInt("id"),
                    rs.getString("title"),
                    rs.getString("content"),
                    rs.getInt("type_id"),
                    rs.getString("type_name")
                );
            }
        } catch (SQLException e) { e.printStackTrace(); }
        return null;
    }

    // ============================================================
    // LẤY DANH SÁCH NOTE TYPES
    // ============================================================
    public List<NoteType> getAllTypes() {
        List<NoteType> types = new ArrayList<>();
        String sql = "SELECT * FROM note_types";
        try (Connection c = DBConnection.getConnection();
             PreparedStatement ps = c.prepareStatement(sql)) {
            ResultSet rs = ps.executeQuery();
            while (rs.next()) {
                types.add(new NoteType(rs.getInt("id"), rs.getString("name")));
            }
        } catch (SQLException e) { e.printStackTrace(); }
        return types;
    }
}

package com.codegym.controller;

import com.codegym.model.Note;
import com.codegym.service.NoteManagement;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;
import java.util.List;

@WebServlet(name = "NoteServlet", urlPatterns = "/notes")
public class NoteServlet extends HttpServlet {

    private NoteManagement noteManagement = new NoteManagement();

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        request.setCharacterEncoding("UTF-8");
        String action = request.getParameter("action");
        if (action == null) action = "";

        switch (action) {
            case "add":
                addNote(request, response);
                break;
            case "changeStore":
                changeStore(request, response);
                break;
            default:
                listNotes(request, response);
                break;
        }
    }

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        request.setCharacterEncoding("UTF-8");
        String action = request.getParameter("action");
        if (action == null) action = "";

        switch (action) {
            case "add":
                showAddForm(request, response);
                break;
            case "detail":
                showDetail(request, response);
                break;
            case "delete":
                deleteNote(request, response);
                break;
            case "search":
                searchNotes(request, response);
                break;
            default:
                listNotes(request, response);
                break;
        }
    }

    private void listNotes(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        List<Note> notes = noteManagement.searchNotes("");
        request.setAttribute("notes", notes);
        request.setAttribute("storeType", noteManagement.getStoreType());
        request.getRequestDispatcher("note/list.jsp").forward(request, response);
    }

    private void searchNotes(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String keyword = request.getParameter("keyword");
        List<Note> notes = noteManagement.searchNotes(keyword);
        request.setAttribute("notes", notes);
        request.setAttribute("keyword", keyword);
        request.setAttribute("storeType", noteManagement.getStoreType());
        request.getRequestDispatcher("note/list.jsp").forward(request, response);
    }

    private void showAddForm(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        request.setAttribute("types", noteManagement.getAllTypes());
        request.setAttribute("storeType", noteManagement.getStoreType());
        request.getRequestDispatcher("note/add.jsp").forward(request, response);
    }

    private void addNote(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String title   = request.getParameter("title");
        String content = request.getParameter("content");
        int typeId     = Integer.parseInt(request.getParameter("typeId"));

        noteManagement.addNote(title, content, typeId);

        request.setAttribute("message", "Đã thêm ghi chú thành công!");
        request.setAttribute("types", noteManagement.getAllTypes());
        request.setAttribute("storeType", noteManagement.getStoreType());
        request.getRequestDispatcher("note/add.jsp").forward(request, response);
    }

    private void showDetail(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        int id = Integer.parseInt(request.getParameter("id"));
        Note note = noteManagement.getNoteById(id);
        request.setAttribute("note", note);
        request.setAttribute("storeType", noteManagement.getStoreType());
        request.getRequestDispatcher("note/detail.jsp").forward(request, response);
    }

    private void deleteNote(HttpServletRequest request, HttpServletResponse response)
            throws IOException {
        int id = Integer.parseInt(request.getParameter("id"));
        noteManagement.addNote(id);   // gọi delete
        response.sendRedirect("notes");
    }

    private void changeStore(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String storeType = request.getParameter("storeType");
        noteManagement.changeNoteStore(storeType);
        response.sendRedirect("notes");
    }
}

⭐ KIẾN THỨC CHÍNH — STRATEGY PATTERN
1. Sơ đồ lớp
text
              ┌─────────────────┐
              │   <<interface>> │
              │       Note      │  ← Strategy
              └────────┬────────┘
                       │
              ┌────────┴─────────┐
              │                  │
       ┌──────▼──────┐    ┌──────▼──────┐
       │   NoteDB    │    │  NoteFile   │
       │ (DB strategy)│    │(File strategy)│
       └─────────────┘    └─────────────┘
              ▲                  ▲
              │                  │
              └──────┬───────────┘
                     │
            ┌────────┴─────────┐
            │  NoteManagement  │  ← Context
            │  changeNoteStore │
            └──────────────────┘
2. Đặc điểm Strategy Pattern
Thành phần	Vai trò	Trong bài
Strategy	Interface chung	Note
Concrete Strategies	Các cách triển khai	NoteDB, NoteFile
Context	Dùng strategy	NoteManagement
3. Ưu điểm
✅ Dễ mở rộng: Thêm NoteXML, NoteJSON chỉ cần implements Note.

✅ Tách biệt logic: NoteManagement không biết chi tiết lưu trữ.

✅ Runtime switching: Đổi changeNoteStore("FILE") bất kỳ lúc nào.

4. Định dạng file notes.txt
text
id|title|typeId|content
1|Họp nhóm|1|Nội dung cuộc họp...
2|Mua sữa|2|Mua 2 hộp sữa tươi
Dùng dấu | làm separator (ít xuất hiện trong nội dung).

Mỗi dòng 1 ghi chú.

ID tự tăng = max(id) + 1.

💡 TRẢ LỜI VẤN ĐÁP NHANH
Câu hỏi	Trả lời
Strategy Pattern là gì?	Cho phép chọn thuật toán/chiến lược tại runtime mà không sửa code client.
Tại sao dùng Strategy cho iNotes?	Vì cần linh hoạt đổi giữa lưu DB và lưu File mà không viết lại logic nghiệp vụ.
changeNoteStore để làm gì?	Đổi strategy hiện tại — chỉ cần đổi đối tượng note từ NoteDB sang NoteFile.
Khác gì với if-else?	If-else hard-code, khó mở rộng. Strategy cho phép plug-in các chiến lược mới dễ dàng.
Nếu muốn thêm lưu XML?	Tạo NoteXML implements Note rồi thêm case trong changeNoteStore.
| trong file là gì?	Separator phân tách các trường: id, title, typeId, content.
Làm sao tìm kiếm trong file?	Đọc toàn bộ file → filter bằng contains(keyword).
Chúc bạn ăn trọn điểm! 🚀
